1. CREATE  src/app/core/services/deployment.service.ts
────────────────────────────────────────
Injectable service (providedIn: 'root') that uses Angular HttpClient.
Base URL: 'http://localhost:8080/api'  (same pattern as the other services in this folder).

Methods:
  getAll():    Observable<Deployment[]>   → GET  /deployments
  getById(id): Observable<Deployment>    → GET  /deployments/{id}
  trigger(repoUrl: string, branch: string): Observable<Deployment>
                                          → POST /deployments
                                            body: { repoUrl, branch }

Deployment interface (define in the same file):
  id: number
  repoUrl: string
  branch: string
  status: 'PENDING' | 'CLONING' | 'BUILDING' | 'DEPLOYING' | 'SUCCESS' | 'FAILED'
  createdAt: string

---- deployments.component.ts ----
Standalone component. Imports: CommonModule, ReactiveFormsModule, HttpClientModule (or use inject).
Use the same patterns as the other pages in this project:
  - constructor-injected services
  - subscribe() in ngOnInit
  - setInterval for polling (clear in ngOnDestroy)
  - loading/error boolean flags

State:
  deployments: Deployment[] = []
  selectedId: number | null = null
  deployLogs: any[] = []        ← logs filtered for the selected deployment
  loading = false
  submitting = false
  error = ''
  form = FormGroup with controls: repoUrl (required), branch (default 'main')

ngOnInit: call loadDeployments(), then setInterval(loadDeployments, 3000)
ngOnDestroy: clearInterval

loadDeployments(): GET /api/deployments via DeploymentService.getAll(),
  assign result to this.deployments, sort by createdAt descending.

onSubmit():
  if form invalid return
  set submitting = true
  call DeploymentService.trigger(repoUrl, branch)
  on next: reset form, call loadDeployments()
  on error: set error message
  finally: submitting = false

selectDeployment(id: number):
  if selectedId === id, set selectedId = null and return (toggle off)
  set selectedId = id
  fetch GET /api/logs via HttpClient (inject it directly here)
  filter the result array keeping only entries where source === 'deployment-worker-' + id
  assign to deployLogs
  sort by timestamp descending

statusColor(status): returns one of these CSS variable strings:
  PENDING   → 'var(--tx-4)'
  CLONING   → 'var(--info)'
  BUILDING  → 'var(--warn)'
  DEPLOYING → 'oklch(0.75 0.15 60)'
  SUCCESS   → 'var(--ok)'
  FAILED    → 'var(--crit)'

levelColor(level): (for log rows)
  INFO  → 'var(--info)'
  WARN  → 'var(--warn)'
  ERROR → 'var(--crit)'
  FATAL → 'var(--crit)'
  DEBUG → 'var(--tx-4)'

                            <span class="mono log-msg">{{ log.message }}</span>
                          </div>
                        }
                      </div>
                    }
                  </div>
                </td>
              </tr>
            }
          }

      </table>
    }
  </div>

</div>
